private suspend fun loginWithRetry(maxRetries: Int = 3) {
    var attempt = 0
    while (attempt < maxRetries) {
        try {
            val result = apiManager.login(username, password)
            if (result.isSuccess) return
            break  // Don't retry credential errors
        } catch (e: IOException) {
            attempt++
