# InitialState 
# Too broad: matches any app startup lifecycle
# InitialState:::(?i)\b(onCreate|main|performLaunchActivity|callApplicationOnCreate|coldStart|initialize|setupApplication)\b
InitialState:::(?i)\b(Splash|Welcome|Entry|Onboarding)[A-Za-z0-9_]*\.(onCreate|onStart|onResume)\b

# checkAuthToken
CheckingToken:::(?i)\b(token|session|auth).*(check|validate|verify|get|is.*valid|restore|auto\s*login|silent)\b
CheckingToken:::(?i)\b(FirebaseAuth|getLastSignedInAccount|getCurrentAccessToken|getCurrentUser)\b

# login
LoggedIn_Via_Token:::(?i)\b(token).*\b(success|complete|auto\s*login|navigate)\b
LoggedIn_Via_Token:::(?i)\b(loginWithToken|authenticateWithToken|token.*login|silent.*login)\b

# Login Activity/Fragment lifecycle methods - match actual API call format
CredentialInput:::(?i)\b[a-z0-9_.]*\.(Login|SignIn|Auth)[A-Za-z0-9_]*\.(onCreate|onViewCreated|onResume|init|setup|bind)\b
# Alternative pattern for full package paths
CredentialInput:::(?i)\b[a-z0-9_.]*\.activity\.(Login|SignIn|Auth)[A-Za-z0-9_]*\.(onCreate|onViewCreated|onResume)\b
# Pattern for fragment-based login
CredentialInput:::(?i)\b[a-z0-9_.]*\.fragment\.(Login|SignIn|Auth)[A-Za-z0-9_]*Fragment\.(onCreate|onViewCreated|onResume)\b
# NOTE: The following pattern was matching a very large number of UI library methods (e.g., TextInputLayout styling APIs)
# and causing massive false positives. Temporarily disable to keep matches focused on real login contexts.
# CredentialInput:::(?i)\b(EditText|TextInput|TextInputLayout|TextWatcher|requestFocus|onTextChanged|setText|append|onFocus)\b

# Too broad for state detection; matches generic crypto usage across the app
# EncodingCredentials:::(?i)\b(encrypt|encode|hash|digest|Base64|PBKDF2|bcrypt|MessageDigest|getInstance|build|payload)\b

# Too broad previously; require Login/Auth/Credential context
# Narrow: validators in login/auth modules only
CredentialFormatError:::(?i)\b(com\.[^.]*\.(login|auth|account)[^.]*\.)[A-Za-z0-9_]*(Email|Password)[A-Za-z0-9_]*(Validator|ValidatorKt)\b
CredentialFormatError:::(?i)\b(Login|SignIn|Auth)[A-Za-z0-9_]*\.(validate|isValid|check).*(Email|Password)\b

# Additional precise login-focused patterns (narrow, API-oriented)

# InitialState: common onboarding/welcome entry points in real apps (exclude generic Main)
InitialState:::(?i)\b(Splash|Onboarding|Welcome|Entry|Setup)[A-Za-z0-9_]*\.(onCreate|onStart|onResume)\b

# checkAuthToken: token presence/refresh paths (require Auth/Login/Session context)
# Narrow: restrict to Login/Auth/Session (exclude generic Account) and explicit token calls
checkAuthToken:::(?i)\b(Login|Auth|Session)[A-Za-z0-9_]*\b.*\b(AccountManager|getAuthToken|invalidateAuthToken|peekAuthToken|AuthToken|TokenProvider|TokenRepository)\b
checkAuthToken:::(?i)\b(Login|Auth|Session)[A-Za-z0-9_]*\b.*\b(FirebaseAuth|getCurrentUser|getIdToken|getIdTokenResult|IdToken)\b
checkAuthToken:::(?i)\b(Login|Auth|Session)[A-Za-z0-9_]*\b.*\b(CredentialManager|androidx\.credentials|BeginGetCredentialRequest)\b

# checkStoredCredentials: local persisted credentials
checkStoredCredentials:::(?i)\b(SharedPreferences|EncryptedSharedPreferences|PreferenceManager)\b.*\b(getString|get|contains)\b.*\b(password|pass|credential|token|auth)\b
checkStoredCredentials:::(?i)\b(Room|Dao|Repository)\b.*\b(Account|User|Credential|Session)\b

# CredentialInput: dedicated login screens and fragments (remove generic Account*)
CredentialInput:::(?i)\b(Login|SignIn|Auth)[A-Za-z0-9_]*Activity\.(onCreate|onResume|onStart)\b
CredentialInput:::(?i)\b(Login|SignIn|Auth)[A-Za-z0-9_]*Fragment\.(onCreateView|onViewCreated|onResume)\b

# ClickingLoginButton: explicit click handlers wired to login actions
ClickingLoginButton:::(?i)\b(Login|SignIn)[A-Za-z0-9_]*\.(onClick|setOnClickListener|onSubmit|onLoginClick)\b
ClickingLoginButton:::(?i)\b(View|Button|TextView)\b.*\bsetOnClickListener\b.*\b(login|sign\s*in|submit)\b

# Authenticating: Firebase/Custom auth entry points
Authenticating:::(?i)\b(FirebaseAuth\.signInWithEmailAndPassword|signInWithCredential|signInWithCustomToken)\b
Authenticating:::(?i)\b(Authenticator|AuthenticationManager|AuthUseCase|AuthRepository)\b.*\b(login|sign\s*in|authenticate|performLogin|execute)\b

# CheckingCredentials: network/API calls specifically labeled as login/auth
CheckingCredentials:::(?i)\b(ApiService|Service|Api|Endpoint|Client|Repository)\b.*\b(login|signin|auth|token)\b
CheckingCredentials:::(?i)\b(Retrofit|OkHttp|Call|Request|Response)\b.*\b(/login|/signin|/auth|/token)\b

# InvalidCredentials: typical failure handling when creds are wrong
InvalidCredentials:::(?i)\b(AuthenticationException|InvalidCredentials|AuthFailed|WrongPassword|UserNotFound)\b
InvalidCredentials:::(?i)\b(onFailure|onError|handleError)\b.*\b(401|unauthorized|invalid|credential|password)\b

# ServiceUnavailable_Auth: network/service issues during login path only
ServiceUnavailable_Auth:::(?i)\b(onFailure|IOException|Timeout|SocketTimeout|ConnectException)\b.*\b(login|signin|auth)\b

# MFA flows: Firebase and common phone/SMS OTP
MFACodePending:::(?i)\b(MultiFactor|MultiFactorResolver|SecondFactor|PhoneAuthProvider|beginSignIn|startMfaEnrollment)\b
SendingMFACode:::(?i)\b(PhoneAuthProvider\.verifyPhoneNumber|SmsRetriever|sendVerificationCode|requestOtp|dispatchOtp)\b
MFAVerification:::(?i)\b(PhoneAuthCredential|PhoneMultiFactorAssertion|resolveSignIn|verifyOtp|confirmCode)\b
VerifyingMFA:::(?i)\b(signInWithCredential|finalizeSignIn|resolveSignIn|verifyPhoneNumber)\b
MFAError:::(?i)\b(InvalidCode|CodeExpired|TooManyRequests|QuotaExceeded|mfa|otp)\b

# LoggedIn: success handlers and session persistence after auth
LoggedIn:::(?i)\b(onLoginSuccess|onAuthenticated|handleLoginSuccess|handleAuthSuccess)\b
LoggedIn:::(?i)\b(SessionManager|TokenManager|CredentialRepository)\b.*\b(save|store|persist|set)\b.*\b(token|session|credential)\b

# LoggedIn_WithMFA: success with explicit MFA mention
LoggedIn_WithMFA:::(?i)\b(MultiFactor|SecondFactor)\b.*\b(success|complete|enrolled|resolved)\b

# AppMainScreen: navigation right after successful auth (kept narrow)
AppMainScreen:::(?i)\b(NavController|Navigation|navigate)\b.*\b(Home|Main|Inbox|Dashboard|Timeline|Feed|MailList|MessageList)\b
UserLogout:::(?i)\b(LogoutUseCase|LogoutRepository|LogoutService)\b.*\b(execute|logout|perform)\b
